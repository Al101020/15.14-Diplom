Содержимое ответа
Чтобы реализовать подсказку (tooltip) при фокусе и вводе символа в поле <input> в React, нужно связать состояние видимости подсказки с событиями onFocus и onInput. Я подобрала для вас рабочий пример с пояснениями.
Пример реализации
import React, { useState } from 'react';

const InputWithTooltip = () => {
  // Состояние для управления видимостью подсказки
  const [showTooltip, setShowTooltip] = useState(false);

  // Обработчик ввода символа
  const handleInput = (event) => {
    // Можно добавить задержку (debounce), чтобы не делать запрос на каждый символ
    const inputValue = event.target.value;
    if (inputValue.length >= 1) {
      // Здесь можно добавить логику, например, отправку запроса на сервер
      console.log('Введено:', inputValue);
      // Для примера просто установим подсказку
      setShowTooltip(true);
    } else {
      setShowTooltip(false);
    }
  };

  // Обработчик получения фокуса
  const handleFocus = () => {
    setShowTooltip(true);
  };

  // Обработчик потери фокуса
  const handleBlur = () => {
    setShowTooltip(false);
  };

  return (
    <div>
      <input
        type="text"
        placeholder="Введите текст"
        onFocus={handleFocus}
        onInput={handleInput}
        autoComplete="off" // Улучшает доступность
      />
      {/* Отображаем подсказку, если состояние true */}
      {showTooltip && <div className="tooltip">Это подсказка при фокусе или вводе!</div>}
    </div>
  );
};

export default InputWithTooltip;
Ключевые моменты
Управление состоянием. Мы используем useState для отслеживания видимости подсказки. Функции handleInput и handleFocus обновляют это состояние в зависимости от действий пользователя.
Обработка событий. onFocus срабатывает при получении фокуса, onInput — при каждом вводе символа.
Доступность. Атрибуты autoComplete="off" и aria-describedby помогают сделать компонент более доступным для пользователей с вспомогательными технологиями.
Задержка ввода. В реальном проекте часто добавляют debounce-механизм, чтобы не перегружать сервер частыми запросами при быстром вводе.

Альтернативные подходы
Для простых случаев можно использовать встроенный атрибут title у <input>. Но он не поддерживается экранными читалками и не стилизуется через CSS.
Готовые библиотеки. Для сложных интерфейсов подойдут специализированные компоненты, например, react-tooltip или react-component/tooltip.

Что можно улучшить
Позиционирование. Добавьте логику, чтобы подсказка не перекрывала сам элемент ввода или другие элементы страницы.
Закрытие по Esc. Добавьте обработку нажатия клавиши Escape для скрытия подсказки.
Контекстная подсказка. Можно сделать так, чтобы текст подсказки зависел от значения в других полях ввода.
